      <input type="number" id="pain-level" min="0" max="10" required />

  <script>
    const painData = {
      0: { emoji: "😐", description: "Tidak ada nyeri.", example: "Sehat tanpa rasa sakit.", tips: "Tetap jaga pola hidup sehat." },
      1: { emoji: "🙂", description: "Nyeri ringan.", example: "Nyeri otot ringan.", tips: "Istirahat cukup dan minum air." },
      2: { emoji: "🙂", description: "Nyeri ringan.", example: "Nyeri yang tidak mengganggu.", tips: "Gunakan kompres hangat." },
      // ... Tambahkan data untuk 3-10 sesuai kebutuhan
      10: { emoji: "😭", description: "Nyeri sangat berat.", example: "Kondisi darurat, nyeri tak tertahankan.", tips: "Hubungi layanan medis segera!" }
    };

    const scaleDiv = document.getElementById('scale-buttons');
    const resultDiv = document.getElementById('result');
    const emojiDiv = document.getElementById('emoji');
    const descriptionDiv = document.getElementById('description');
    const exampleDiv = document.getElementById('example');
    const tipsDiv = document.getElementById('tips');

    // Membuat tombol untuk skala nyeri
    for (let i = 0; i <= 10; i++) {
      const btn = document.createElement('button');
      btn.textContent = i;
      btn.onclick = () => showPainInfo(i);
      scaleDiv.appendChild(btn);
    }

    function showPainInfo(level) {
      const data = painData[level];
      emojiDiv.textContent = data.emoji;
      descriptionDiv.innerHTML = `<strong>Deskripsi:</strong> ${data.description}`;
      exampleDiv.innerHTML = `<strong>Contoh Kondisi:</strong> ${data.example}`;
      tipsDiv.innerHTML = `<strong>Tips:</strong> ${data.tips}`;
      resultDiv.style.display = 'block';
      window.scrollTo({ top: resultDiv.offsetTop, behavior: 'smooth' });
    }

    // Simpan data dari form pencatatan nyeri
    document.getElementById('pain-form').addEventListener('submit', (e) => {
      e.preventDefault();
      const level = document.getElementById('pain-level').value;
      const location = document.getElementById('pain-location').value;
      const notes = document.getElementById('pain-notes').value;

      alert(`Catatan Nyeri Tersimpan:\nTingkat: ${level}\nLokasi: ${location}\nCatatan: ${notes}`);
    });
  </script>
